}

岗位要求：掌握 Python、SQL 和 Excel，具备数据分析及数据可视化能力。
能够进行跨部门沟通，具备英语阅读能力。"""


def contains(text, keyword):
    """英文关键词忽略大小写，避免将 JavaScript 匹配为 Java。"""
    if keyword.isascii():
        pattern = (
            r"(?<![a-z0-9_])"
            + re.escape(keyword)
            + r"(?![a-z0-9_])"
        )
        return bool(re.search(pattern, text, re.I))
    return keyword in text


def extract(text):
    """提取能力组，并保留命中的关键词作为依据。"""
    return {
        name: next(
            (alias for alias in aliases if contains(text, alias)),
            None,
        )
        for name, aliases in SKILLS.items()
        if any(contains(text, alias) for alias in aliases)
    }


def analyze(resume, jd):
    """分析简历对 JD 中已支持能力词的覆盖情况。"""
    if not resume.strip() or not jd.strip():
        raise ValueError("请同时填写简历文本和岗位 JD。")

    if max(len(resume), len(jd)) > 20000:
        raise ValueError("每项文本最多支持 20000 个字符。")

    required = extract(jd)
    supplied = extract(resume)

    matched = [skill for skill in required if skill in supplied]
    missing = [skill for skill in required if skill not in supplied]

    evidence = [
        {
            "能力": skill,
            "JD关键词": required[skill],
            "简历关键词": supplied[skill],
        }
        for skill in matched
    ]

    suggestions = [
        (
            f"{skill}：若确有相关经历，可补充真实的任务、使用方法及可核实结果；"
            "没有经历时，请先学习或完成练习项目，不要声称已经掌握。"
        )
        for skill in missing
    ]
    suggestions += [
        (
            "对已有经历使用“任务—行动—结果”结构；"
            "只有确有记录时才补充数字，不虚构公司、任职时间、职责或业绩。"
        ),
        "课程或练习项目请明确标注为课程或练习，不包装成工作经历。",
    ]

    return {
        "score": (
            round(len(matched) / len(required) * 100)
            if required
            else None
        ),
        "required": list(required),
        "matched": matched,
        "missing": missing,
        "evidence": evidence,
        "suggestions": suggestions,
    }


def report(result):
    """生成可下载的 Markdown 报告，不附带完整原始输入。"""
    score = (
        f'{result["score"]}/100'
        if result["score"] is not None
        else "无法评分（未识别到支持的能力词）"
    )

    lines = [
        "# 求职匹配参考报告",
        "",
        f"参考评分：{score}",
        (
            "规则：JD 中识别到的能力组等权计分，"
            "分数 = 简历出现的能力组数 / JD 能力组数 × 100，四舍五入。"
        ),
        (
            "仅检查词语出现，不验证能力、经历真实性、否定语义、年限或学历，"
            "不预测录用概率。"
        ),
        "缺失表示未找到关键词，不代表没有该能力；重复词不加分。",
        "",
        "## 匹配依据",
    ]

    lines += [
        (
            f'- {item["能力"]}：JD 出现「{item["JD关键词"]}」，'
            f'简历出现「{item["简历关键词"]}」。'
        )
        for item in result["evidence"]
    ] or ["无匹配项。"]

    lines += [
        "",
        "## 未体现的能力",
        "、".join(result["missing"]) or "无；仍需人工核实要求是否满足。",
        "",
        "## 诚实的优化建议",
    ]

    lines += [
        f"- {suggestion}"
        for suggestion in result["suggestions"]
    ]

    return "\n".join(lines) + "\n"
